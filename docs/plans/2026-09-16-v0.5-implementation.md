**Goal:** 让 V0 变得可验证、可观测、可自证存活 —— 交付回放器、`--report` 观测层与 `run_events` 健康信号。

**Architecture:** 三种协作分工。`store/` 交出原始行（新增只读查询）；`report/` 把行聚合成意思（纯函数，无 IO）；`replay/` 用合成数据驱动**真实的 `Scheduler`**，因此验证的是生产链路本身而非平行复刻品。健康信号不新建心跳表 —— `ActivityReader.read()` 永不抛异常，`evaluations.data_status` 已是一份逐 tick 的断连日志；只需补上两处"静默不落行"的缺口。

**Tech Stack:** Python 3.12、uv、pytest、仅标准库（`sqlite3` / `tomllib` / `argparse` / `dataclasses` / `datetime`）。运行期第三方依赖保持 0。

## Global Constraints

以下约束适用于**每一个**任务，不再逐条重复：

- **运行期第三方依赖保持 0。** `pyproject.toml` 的 `dependencies = []` 不得改动。`dev` 组只允许 `pytest`。
- **`uv run pytest -q` 必须始终通过。** 当前基线 **154 passed**。改动既有测试断言（如 `user_version == 3` → `4`）是允许的，但不得删除测试。
- **Conventional Commits**，`scope` 取组件名：`config` / `reader` / `state` / `gate` / `store` / `notify` / `outcome` / `scheduler` / `report` / `replay`。描述用中文。
- **`--report` 严格只读。** 不得出现 `INSERT` / `UPDATE` / `DELETE` / `CREATE`。`Store` 新增方法一律以 `list_` 开头以便评审时一眼确认。
- **`run_events` 绝不写进 `interventions` 表。** 运维信号与干预信号必须分离，否则会占用 `daily_cap` 配额、消耗 `cooldown`，并在 `outcomes` 里混进不是干预的数据。
- **健康告警绝不弹窗。** 只经由 `--report` 呈现。
- **不新增窗口标题的持久化。** report 可以打印标题到 stdout，但不得落库。
- **`entries_minutes` 不参与 `classify` 与任何闸门判定。** 状态结论必须与 V0 逐位一致。
- **`SCHEMA_VERSION` 递增到 4**，迁移沿用 `_apply_incremental_migrations()` 的逐列补齐机制，老库原地升级。
- **配置启动即校验，非法值直接失败，绝不静默降级。**
- **`data_status` 与 `run_events.kind` 都是封闭枚举**，枚举外的值不得被当作正常值处理。

```python
    def run_forever(self) -> None:  # pragma: no cover - 常驻路径靠手动验证
        interval = timedelta(minutes=self._config.schedule.evaluate_every_minutes)
        last: datetime | None = None
        while True:
            now = self._clock.now()
            if last is None or now - last >= interval:
                self.run_tick_guarded(now)
                last = now
            time.sleep(self._config.schedule.tick_seconds)

    def run_tick_guarded(self, now: datetime) -> TickReport | None:
        """常驻循环的单轮：异常必须落库，不能只留在日志里。

        写事件本身若也失败（磁盘满、库损坏），只能退回日志 —— 此时观测退化为
        「看起来像进程死了」，spec §7.6 已记录该边界。
        """
        try:
            return self.run_once()
        except Exception as exc:  # noqa: BLE001 - 常驻进程不能因为单轮失败就退出
            log.exception("本轮评估失败，跳过")
            try:
                self._store.insert_run_event(
                    now, "tick_error", f"{type(exc).__name__}: {exc}"[:200]
                )
            except Exception:  # noqa: BLE001
                log.exception("写入 tick_error 运行事件失败")
            return None
```

> 把单轮逻辑抽成 `run_tick_guarded` 是为了让它**可测**：常驻循环本身带 `time.sleep`，无法在测试里直接跑。抽出后测试可以只调这一轮。

```python
    # ── 只读查询（report 用） ────────────────────────────────
    # 一律以 list_ 开头，便于评审时确认 report 路径不写库。
    # 时间过滤用 SQL 字符串比较：_iso() 统一转本地时区后序列化，

    def list_evaluations(self, since: datetime | None = None) -> list[sqlite3.Row]:
        if since is None:
            return self._conn.execute("SELECT * FROM evaluations ORDER BY at").fetchall()
        return self._conn.execute(
            "SELECT * FROM evaluations WHERE at >= ? ORDER BY at", (_iso(since),)
        ).fetchall()

    def list_interventions(self, since: datetime | None = None) -> list[sqlite3.Row]:
        if since is None:
            return self._conn.execute("SELECT * FROM interventions ORDER BY at").fetchall()
        return self._conn.execute(
            "SELECT * FROM interventions WHERE at >= ? ORDER BY at", (_iso(since),)
        ).fetchall()

    def list_outcomes(self, since: datetime | None = None) -> list[sqlite3.Row]:
        if since is None:
            return self._conn.execute("SELECT * FROM outcomes ORDER BY checked_at").fetchall()
        return self._conn.execute(
            "SELECT * FROM outcomes WHERE checked_at >= ? ORDER BY checked_at", (_iso(since),)
        ).fetchall()

    def list_run_events(self, since: datetime | None = None) -> list[sqlite3.Row]:
        if since is None:
            return self._conn.execute("SELECT * FROM run_events ORDER BY at").fetchall()
        return self._conn.execute(
            "SELECT * FROM run_events WHERE at >= ? ORDER BY at", (_iso(since),)
```

```python
def build_overview(evaluations, run_events, *, evaluate_every_minutes, gap_threshold_minutes):
    moments = [_parse(r["at"]) for r in evaluations]
    events = tuple(RunEvent(_parse(r["at"]), r["kind"], r["detail"]) for r in run_events)
    if not moments:
        return Overview(None, None, 0, 0, 0.0, ())

    first, last = moments[0], moments[-1]
    span_minutes = (last - first).total_seconds() / 60
    expected = int(span_minutes // evaluate_every_minutes) + 1 if span_minutes > 0 else 1
    coverage = (len(moments) / expected) if expected else 0.0

    gaps = []
    for prev, cur in zip(moments, moments[1:]):
        minutes = (cur - prev).total_seconds() / 60
        if minutes > gap_threshold_minutes:
            inside = tuple(e for e in events if prev <= e.at <= cur)
            gaps.append(ContinuityGap(prev, cur, round(minutes, 1), inside))
    return Overview(first, last, len(moments), expected, round(coverage, 3), tuple(gaps))
```

```python
def build_gate_breakdown(evaluations, *, ratio_buckets: int = 10):
    blocked: Counter[str] = Counter()
    histogram: Counter[str] = Counter()
    candidates: list[GateBlock] = []
    for row in evaluations:
        trace = _parse_trace(row["gate_trace"])
        first_failed = next((g for g in trace if not g["passed"]), None)
        if first_failed is not None:
            blocked[first_failed["name"]] += 1
            # state_min 通过、却被后续闸门挡下 —— 这才是「该提醒但没提醒」
            if first_failed["name"] != "state_min":
                candidates.append(
                    GateBlock(_parse(row["at"]), first_failed["name"],
                              first_failed["value"], first_failed["threshold"])
                )
        if row["total_active_minutes"] > 0:
            histogram[_ratio_bucket(row["ent_ratio"], ratio_buckets)] += 1
    return GateBreakdown(
        blocked_by=tuple(sorted(blocked.items(), key=lambda kv: -kv[1])),
        ratio_histogram=tuple(sorted(histogram.items())),
        state_min_passed_then_blocked=tuple(candidates),
    )
```

```python
def find_leak_anchors(evaluations, *, min_active_minutes, min_unclassified_ratio):
    anchors = []
    for row in evaluations:
        total = row["total_active_minutes"]
        if total < min_active_minutes:
            continue
        entries = row["entries_minutes"]
        # 先扣掉「明细缺失」：那是数据没取到，不是规则漏判。
        # 旧库（v4 之前）该列为 DEFAULT 0，此时 missing 等于 total，
        # unclassified 归零，锚点自然不触发 —— 不伪造数据。
        missing = max(total - entries, 0.0)
        unclassified = max(entries - (row["ent_minutes"] + row["gray_minutes"]
                                      + row["work_minutes"]), 0.0)
        if total <= 0:
            continue
        ratio = unclassified / total
        if ratio >= min_unclassified_ratio:
            anchors.append(LeakAnchor(
                at=_parse(row["at"]), total_active_minutes=total,
                ent_minutes=row["ent_minutes"], gray_minutes=row["gray_minutes"],
                work_minutes=row["work_minutes"],
                unclassified_minutes=round(unclassified, 2),
                unclassified_ratio=round(ratio, 3),
                missing_detail_minutes=round(missing, 2),
            ))
    return tuple(anchors)
```

`build_verdict_breakdown` / `build_intervention_breakdown` / `build_outcome_breakdown` / `build_trace` 按 **Interfaces** 里的签名实现：

- `build_outcome_breakdown`：`by_response` 的键取 `interventions.user_response`，`None` 用字符串 `"none"`（保持 dataclass 可比较、可渲染）；`ent_before_mean` / `ent_after_mean` 只统计 `outcome != "no_data"` 的行。
- `build_trace`：取 `at` 前后各 `window_ticks // 2` 轮，`at` 本身若不在行里则取最接近的不晚于它的那一行。
- `_parse_trace(text)`：`json.loads` 失败返回 `()`，由调用方计入"闸门数据损坏"（渲染层显示，不抛异常）。

```python
"""report 的渲染。纯函数：输入 dataclass，输出字符串，不碰 IO。"""

from __future__ import annotations

import json
from dataclasses import asdict

from statesense.report.models import ReportData


def _gap_verdict(gap) -> str:
    """缺口旁边写清它是什么 —— 这正是本版本要消灭的二义。"""
    if gap.events:
        kinds = "、".join(f"{e.kind}({e.detail})" for e in gap.events)
        return f"有运行事件记录：{kinds}"
    return "无运行事件记录 → 进程当时不在运行"


def render_text(data: ReportData) -> str:
    lines: list[str] = []
    ov = data.overview

    lines.append("概览")
    if ov.evaluations == 0:
        lines.append("  区间内无任何评估记录。")
        lines.append("  结论：进程从未跑起来，或库路径不对。")
        return "\n".join(lines)

    lines.append(f"  区间        {ov.first_at} → {ov.last_at}")
    lines.append(f"  评估轮数    {ov.evaluations}（应约 {ov.expected_evaluations}）")
    lines.append(f"  覆盖率      {ov.coverage:.1%}")

    if ov.gaps:
        lines.append(f"  缺口        {len(ov.gaps)} 处")
        for gap in ov.gaps:
            lines.append(f"    {gap.start} → {gap.end}   {gap.minutes} 分钟")
            lines.append(f"      {_gap_verdict(gap)}")
    else:
        lines.append("  缺口        无")

    skipped_pct = (data.verdicts.skipped / data.verdicts.total) if data.verdicts.total else 0.0
    if skipped_pct:
        lines.append(f"  skipped     {data.verdicts.skipped} 轮（{skipped_pct:.1%}）")

    lines.append("")
    if ov.gaps:
        lines.append(f"结论：数据有过 {len(ov.gaps)} 处中断，看上方每处的成因。")
    elif skipped_pct >= 0.2:
        lines.append(f"结论：连续性正常，但有 {skipped_pct:.1%} 的轮次取不到数据。")
    else:
        lines.append("结论：数据可信。")
    return "\n".join(lines)
```

```python
def _parse_since(text: str, now: datetime) -> datetime:
    """支持 `7d` / `24h` / ISO 时刻。解析失败按配置错误处理，不猜。"""
    text = text.strip()
    if text.endswith("d") and text[:-1].isdigit():
        return now - timedelta(days=int(text[:-1]))
    if text.endswith("h") and text[:-1].isdigit():
        return now - timedelta(hours=int(text[:-1]))
    try:
        moment = datetime.fromisoformat(text)
    except ValueError as exc:
        raise ConfigError(f"--since 无法解析：{text!r}（支持 7d / 24h / ISO 时刻）") from exc
    return moment if moment.tzinfo else moment.astimezone()
```

```python
class ScriptedReader:
    """满足 ActivitySource 的合成数据源。"""

    def __init__(self, scenario: Scenario, window_minutes: int) -> None:
        self._scenario = scenario
        self._window = window_minutes
        self.calls: list[tuple[datetime, datetime]] = []

    def read(self, start, end, window_minutes, captured_at) -> ActivitySnapshot:
        self.calls.append((start, end))
        return snapshot_at(self._scenario, captured_at, window_minutes)


def run_scenario(scenario, config, *, start):
    """用 FrozenClock + 脚本 reader 驱动**真实的 Scheduler**。

    不重跑判定逻辑 —— 被验证的必须是生产链路本身。
    """
    store = Store(config.store_path)
    store.migrate()
    clock = FrozenClock(start)
    reader = ScriptedReader(scenario, config.schedule.window_minutes)
    scheduler = Scheduler(
        config=config, clock=clock, reader=reader, store=store,
        notifier=RecordingNotifier(clock=clock),
    )
    return store, scheduler
```
